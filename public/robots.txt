# Default rules. Search engines and live answer engines (ClaudeBot,
# PerplexityBot, Perplexity-User, OAI-SearchBot) are not named in the block
# group below, so they fall through here and may crawl and cite public pages.
User-agent: *
Disallow: /admin
Disallow: /api
Disallow: /imprint
Disallow: /privacy

# Share-preview cards under /og/ are generated on demand for social scrapers,
# so a preview shows up when someone posts a link. SEO and index crawlers fetch
# the card every page points at without ever displaying one, so they are asked
# to skip the cards while the pages themselves stay crawlable. Preview fetchers
# (Twitterbot, facebookexternalhit, Slackbot, Discordbot, TelegramBot,
# LinkedInBot) are deliberately not in this group. The group repeats the
# default disallows, because a named group replaces the wildcard group rather
# than adding to it.

User-agent: MJ12bot

User-agent: Amazonbot

Disallow: /admin
