# Default rules. Search engines and live answer engines (ClaudeBot,
# PerplexityBot, Perplexity-User, OAI-SearchBot) are not named in the block
# group below, so they fall through here and may crawl and cite public pages.
User-agent: *
Disallow: /admin
Disallow: /api
Disallow: /imprint
Disallow: /privacy

# Share-preview cards under /og/ are generated on demand for social scrapers,
# so a preview shows up when someone posts a link. Amazonbot crawls every page
# and fetches the card each page points at, without ever displaying one, so it
# is asked to skip them while the pages themselves stay crawlable. This group
# repeats the default disallows, because a named group replaces the wildcard
# group rather than adding to it.
User-agent: Amazonbot
Disallow: /og/
Disallow: /admin
Disallow: /api
Disallow: /imprint
Disallow: /privacy

# AI training crawlers: blocked. Public deliberation should be discoverable in
# answer engines, but the content is not offered up for model training. The
# live citing bots above are deliberately kept out of this group, and so is
# Meta-ExternalFetcher, Meta's on-demand link/preview fetcher. Only the
# training crawler Meta-ExternalAgent is blocked below.
User-agent: GPTBot
User-agent: Claude-Web
User-agent: anthropic-ai
User-agent: CCBot
User-agent: Google-Extended
User-agent: Applebot-Extended
User-agent: meta-externalagent
Disallow: /
